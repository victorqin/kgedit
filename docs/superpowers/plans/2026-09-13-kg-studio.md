# KG Studio Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 `KG Studio.dc.html` 设计稿实现为可对接真实后端的知识图谱编辑器，数据源从内存原型换成接口，并可在 mock 与真实后端之间一键切换。

**Architecture:** 单页应用，左右两个以 Start / End 节点为中心的 N 跳子图面板，下方连接关系列表。所有数据经 axios 走统一信封的 REST 接口；mock 模式由 MSW 在 Service Worker 层拦截，业务代码零改动。状态集中在单个 Zustand store（五个 slice），读操作靠请求序号防竞态，写操作靠全局蒙层串行化。

**Tech Stack:**（安装时 npm 解析到的实际版本，已逐项验证 API）Vite 8.3 · React 19.3 · TypeScript 6.0 · **Ant Design 6.6**（非计划最初假设的 5.x；`theme.darkAlgorithm` 与 `antd/locale/{en_US,zh_CN,zh_TW}` 均在，原生支持 React 19，故不需要 `@ant-design/v5-patch-for-react-19`）· @antv/g6 5.1.1（已确认内置 `html` 节点、`quadratic` 边、`d3-force` 布局、`Graph extends EventEmitter`）· Zustand 5.0 + immer 11 · axios 1.20 · MSW 2.15 · react-router 7.18 · react-i18next · Vitest 5（`test` 配置需从 `vitest/config` 导入 `defineConfig`）· Playwright

**Spec:** `docs/superpowers/specs/2026-09-13-kg-studio-design.md`

## Global Constraints

- 包管理器 **npm**（环境已有 npm 11.5.1 / Node 24.6.0；pnpm 不可用）
- Ant Design **6.x** API（`ConfigProvider` + `theme.darkAlgorithm`）
- 界面默认语言 **en-US**，文案全部走 i18n key；另备 `zh-CN`、`zh-TW`
- 单文件 200–400 行为常态，**800 行为硬上限**
- **不可变更新**：禁止就地修改对象，store 内一律经 immer draft 或返回新对象
- 图约束：`UNIQUE(source, target, predicate)`；**禁止自环**（服务端 422）
- hops：整数、`min=1`、**无上限**，由服务端 `nodeCap`（默认 200）兜底
- 删除为**物理删除**，无软删无撤销，仅二次确认
- **不做并发控制**，后写覆盖，实体不带 `version` 字段
- 写操作：二次确认 → 全局蒙层（延迟 250ms 显示）→ 串行 → 并行刷新受影响的读 → 蒙层撤下
- 读操作：不上全局蒙层，仅面板内 Spin，靠请求序号丢弃过期响应
- 强调色：左 `#4ade80`，右 `#60a5fa`，底色 `#0b0f15`，字体 IBM Plex Sans / IBM Plex Mono
- 动画只走 `transform` / `opacity`
- 每个任务结束时提交一次，提交信息用 `<type>: <description>` 约定式格式

```ts
import i18n from 'i18next'
import { initReactI18next } from 'react-i18next'
import LanguageDetector from 'i18next-browser-languagedetector'
import enUS from '@/locales/en-US.json'
import zhCN from '@/locales/zh-CN.json'
import zhTW from '@/locales/zh-TW.json'
import antdEnUS from 'antd/locale/en_US'
import antdZhCN from 'antd/locale/zh_CN'
import antdZhTW from 'antd/locale/zh_TW'
import type { Locale } from 'antd/locale'

export const SUPPORTED_LNGS = ['en-US', 'zh-CN', 'zh-TW'] as const
export type Lng = (typeof SUPPORTED_LNGS)[number]

const ANTD_LOCALES: Record<Lng, Locale> = {
  'en-US': antdEnUS, 'zh-CN': antdZhCN, 'zh-TW': antdZhTW,
}

export const antdLocaleFor = (lng: string): Locale =>
  ANTD_LOCALES[(lng as Lng)] ?? antdEnUS

i18n.use(LanguageDetector).use(initReactI18next).init({
  resources: {
    'en-US': { translation: enUS },
    'zh-CN': { translation: zhCN },
    'zh-TW': { translation: zhTW },
  },
  fallbackLng: 'en-US',
  supportedLngs: SUPPORTED_LNGS,
  detection: { order: ['localStorage', 'navigator'], caches: ['localStorage'] },
  interpolation: { escapeValue: false },
})

export default i18n
```

- [ ] **Step 4: 写 db.ts**

要点：状态放闭包内，每个方法返回**深拷贝**，避免调用方改到内部数据（对应全局约束的不可变要求）。

```ts
import type {
  KgNode, KgEdge, LinkDetail, GraphPayload, TreeResponse, TreeNode,
  PathResponse, PathSegment, Taxonomy, Stats, NodeInput, LinkInput,
  DeleteNodeResult, DeleteLinkResult,
} from '@/api/types'
import { ERR } from '@/api/types'
import { SEED_NODES, SEED_EDGES } from './seed'

export class DbError extends Error {
  constructor(public code: number, message: string) { super(message); this.name = 'DbError' }
}

export const DEFAULT_NODE_CAP = 200
const clone = <T,>(v: T): T => JSON.parse(JSON.stringify(v))

export function createDb(
  seedNodes: KgNode[] = SEED_NODES,
  seedEdges: KgEdge[] = SEED_EDGES,
) {
  let nodes: KgNode[] = clone(seedNodes)
  let edges: KgEdge[] = clone(seedEdges)
  let seq = seedEdges.length

  const nodeOf = (id: string) => nodes.find((n) => n.id === id)
  const mustNode = (id: string) => {
    const n = nodeOf(id)
    if (!n) throw new DbError(ERR.NOT_FOUND, `Node ${id} not found`)
    return n
  }
  const degreeOf = (id: string) =>
    edges.filter((e) => e.source === id || e.target === id).length

  const endpoint = (id: string) => ({ id, label: nodeOf(id)?.label ?? id })
  const detail = (e: KgEdge): LinkDetail => ({
    id: e.id, source: endpoint(e.source), target: endpoint(e.target),
    label: e.label, predicate: e.predicate, directed: true, props: e.props,
    createdAt: new Date(0).toISOString(), updatedAt: new Date(0).toISOString(),
  })

  /** 无向 BFS 分层展开；超出 cap 时按层截断，但中心节点永远保留 */
  function neighborhood(centerId: string, hops: number, cap = DEFAULT_NODE_CAP): GraphPayload {
    mustNode(centerId)
    const dist: Record<string, number> = { [centerId]: 0 }
    const ordered = [centerId]
    let frontier = [centerId]
    for (let h = 0; h < hops; h++) {
      const next: string[] = []
      for (const id of frontier) {
        for (const e of edges) {
          const nb = e.source === id ? e.target : e.target === id ? e.source : null
          if (nb && dist[nb] === undefined) { dist[nb] = h + 1; ordered.push(nb); next.push(nb) }
        }
      }
      if (!next.length) break
      frontier = next
    }
    const totalNodes = ordered.length
    const kept = ordered.slice(0, Math.max(1, cap))
    const keptSet = new Set(kept)
    return {
      nodes: kept.map((id) => clone(mustNode(id))),
      edges: clone(edges.filter((e) => keptSet.has(e.source) && keptSet.has(e.target))),
      meta: { centerId, hops, truncated: totalNodes > kept.length, nodeCap: cap, totalNodes },
    }
  }

  function tree(q: string, limit = 50): TreeResponse {
    const needle = q.trim().toLowerCase()
    let list = needle
      ? nodes.filter((n) =>
          n.label.toLowerCase().includes(needle) ||
          n.type.toLowerCase().includes(needle) ||
          n.domain.toLowerCase().includes(needle))
      : [...nodes].sort((a, b) => degreeOf(b.id) - degreeOf(a.id)).slice(0, limit)

    const byDomain = new Map<string, Map<string, KgNode[]>>()
    for (const n of list) {
      const d = n.domain || 'Uncategorized'
      const t = n.type || '(no type)'
      if (!byDomain.has(d)) byDomain.set(d, new Map())
      const types = byDomain.get(d)!
      types.set(t, [...(types.get(t) ?? []), n])
    }

    const treeOut: TreeNode[] = [...byDomain.entries()].sort(([a], [b]) => a.localeCompare(b))
      .map(([domain, types]) => {
        const children: TreeNode[] = [...types.entries()].sort(([a], [b]) => a.localeCompare(b))
          .map(([type, ns]) => ({
            key: `d:${domain}|t:${type}`, label: type, count: ns.length,
            children: ns.map((n) => ({
              key: `n:${n.id}`, nodeId: n.id, label: n.label,
              meta: n.desc.length > 22 ? `${n.desc.slice(0, 21)}…` : n.desc,
            })),
          }))
        return {
          key: `d:${domain}`, label: domain.toUpperCase(),
          count: children.reduce((a, c) => a + c.children!.length, 0),
          children,
        }
      })
    return { total: list.length, tree: treeOut }
  }

  /**
   * 无向 BFS 求最短路径，但记录每段是否逆着边方向走。
   * 设计稿原型遗漏了这一点，导致逆向段的箭头方向是错的。
   */
  function path(from: string, to: string): PathResponse {
    const direct = edges
      .filter((e) => (e.source === from && e.target === to) || (e.source === to && e.target === from))
      .map(detail)
    if (!nodeOf(from) || !nodeOf(to)) return { found: false, segments: [], directLinks: direct }
    if (from === to) return { found: true, segments: [{ node: endpoint(from) }], directLinks: direct }

    const prev: Record<string, { from: string; edge: KgEdge; reversed: boolean }> = {}
    const seen = new Set([from])
    let queue = [from]
    while (queue.length) {
      const next: string[] = []
      for (const id of queue) {
        for (const e of edges) {
          const isOut = e.source === id
          const nb = isOut ? e.target : e.target === id ? e.source : null
          if (!nb || seen.has(nb)) continue
          seen.add(nb)
          prev[nb] = { from: id, edge: e, reversed: !isOut }
          if (nb === to) {
            const segs: PathSegment[] = []
            let cur = to
            while (cur !== from) {
              const p = prev[cur]
              segs.unshift({
                node: endpoint(cur),
                edge: { id: p.edge.id, label: p.edge.label, reversed: p.reversed },
              })
              cur = p.from
            }
            segs.unshift({ node: endpoint(from) })
            return { found: true, segments: segs, directLinks: direct }
          }
          next.push(nb)
        }
      }
      queue = next
    }
    return { found: false, segments: [], directLinks: direct }
  }

  function assertLinkable(source: string, target: string, predicate: string, ignoreId?: string) {
    if (source === target) throw new DbError(ERR.SELF_LOOP, 'A relation cannot point a node at itself')
    mustNode(source); mustNode(target)
    const dup = edges.some((e) =>
      e.id !== ignoreId && e.source === source && e.target === target && e.predicate === predicate)
    if (dup) throw new DbError(ERR.DUPLICATE, 'This relation already exists')
  }

  return {
    stats: (): Stats => ({ nodeCount: nodes.length, linkCount: edges.length }),
    neighborhood, tree, path,

    taxonomy: (): Taxonomy => ({
      domains: [...new Set(nodes.map((n) => n.domain).filter(Boolean))].sort(),
      types: [...new Set(nodes.map((n) => n.type).filter(Boolean))].sort(),
    }),

    nodeLinks: (id: string): LinkDetail[] => {
      mustNode(id)
      return edges.filter((e) => e.source === id || e.target === id).map(detail)
    },

    link: (id: string): LinkDetail => {
      const e = edges.find((x) => x.id === id)
      if (!e) throw new DbError(ERR.NOT_FOUND, `Link ${id} not found`)
      return detail(e)
    },

    createNode: (input: NodeInput): KgNode => {
      const n: KgNode = { id: `n${Date.now().toString(36)}${seq++}`, ...input }
      nodes = [...nodes, n]
      return clone(n)
    },

    updateNode: (id: string, patch: Partial<NodeInput>): KgNode => {
      mustNode(id)
      nodes = nodes.map((n) => (n.id === id ? { ...n, ...patch } : n))
      return clone(mustNode(id))
    },

    deleteNode: (id: string): DeleteNodeResult => {
      mustNode(id)
      const deletedLinkIds = edges.filter((e) => e.source === id || e.target === id).map((e) => e.id)
      nodes = nodes.filter((n) => n.id !== id)
      edges = edges.filter((e) => e.source !== id && e.target !== id)
      return { deletedNodeId: id, deletedLinkIds }
    },

    createLink: (input: LinkInput): LinkDetail => {
      const predicate = input.predicate ?? input.label
      assertLinkable(input.source, input.target, predicate)
      const e: KgEdge = {
        id: `e${Date.now().toString(36)}${seq++}`,
        source: input.source, target: input.target,
        label: input.label, predicate, directed: true,
      }
      edges = [...edges, e]
      return detail(e)
    },

    updateLink: (id: string, patch: { label?: string; predicate?: string }): LinkDetail => {
      const e = edges.find((x) => x.id === id)
      if (!e) throw new DbError(ERR.NOT_FOUND, `Link ${id} not found`)
      const predicate = patch.predicate ?? patch.label ?? e.predicate
      assertLinkable(e.source, e.target, predicate, id)
      edges = edges.map((x) => (x.id === id ? { ...x, ...patch, predicate } : x))
      return detail(edges.find((x) => x.id === id)!)
    },

    reverseLink: (id: string): LinkDetail => {
      const e = edges.find((x) => x.id === id)
      if (!e) throw new DbError(ERR.NOT_FOUND, `Link ${id} not found`)
      assertLinkable(e.target, e.source, e.predicate, id)
      edges = edges.map((x) => (x.id === id ? { ...x, source: x.target, target: x.source } : x))
      return detail(edges.find((x) => x.id === id)!)
    },

    deleteLink: (id: string): DeleteLinkResult => {
      if (!edges.some((e) => e.id === id)) throw new DbError(ERR.NOT_FOUND, `Link ${id} not found`)
      edges = edges.filter((e) => e.id !== id)
      return { deletedLinkId: id }
    },
  }
}

export type KgDb = ReturnType<typeof createDb>
```

```ts
/**
 * 读操作的竞态守卫。用户快速切换中心节点时，先发的请求可能后返回，
 * 会把新数据覆盖成旧的。发请求前取号，回来时对不上就丢弃。
 */
export function createSeq() {
  const counters = new Map<string, number>()
  return {
    next(key: string): number {
      const v = (counters.get(key) ?? 0) + 1
      counters.set(key, v)
      return v
    },
    isCurrent(key: string, seq: number): boolean {
      return counters.get(key) === seq
    },
  }
}

export const seq = createSeq()
```

```ts
import type { GraphPayload, PathResponse, Stats } from '@/api/types'
import { getNeighborhood, getPath, getStats } from '@/api/graph'
import { ApiError } from '@/api/client'
import { seq } from '@/lib/requestSeq'
import type { Side } from '@/theme/antdTheme'
import type { SliceCreator } from '../useKgStore'

export interface GraphSlice {
  startId: string | null
  endId: string | null
  hops: number
  leftSub: GraphPayload | null
  rightSub: GraphPayload | null
  path: PathResponse | null
  stats: Stats | null
  loading: { L: boolean; R: boolean; path: boolean }
  error: { L: string | null; R: string | null }
  setHops: (n: number) => void
  setCenter: (side: Side, id: string) => Promise<void>
  loadSide: (side: Side) => Promise<void>
  loadPath: () => Promise<void>
  loadStats: () => Promise<void>
  refreshAll: () => Promise<void>
}

const centerKey = (side: Side) => (side === 'L' ? 'startId' : 'endId') as const
const subKey = (side: Side) => (side === 'L' ? 'leftSub' : 'rightSub') as const

export const createGraphSlice: SliceCreator<GraphSlice> = (set, get) => ({
  startId: null, endId: null, hops: 2,
  leftSub: null, rightSub: null, path: null, stats: null,
  loading: { L: false, R: false, path: false },
  error: { L: null, R: null },

  // 整数、下限 1、无上限；上限交由服务端 nodeCap 兜底
  setHops: (n) => {
    const next = Number.isFinite(n) ? Math.max(1, Math.floor(n)) : 1
    set((s) => { s.hops = next })
    void get().loadSide('L')
    void get().loadSide('R')
  },

  setCenter: async (side, id) => {
    set((s) => { s[centerKey(side)] = id })
    await Promise.all([get().loadSide(side), get().loadPath()])
  },

  loadSide: async (side) => {
    const centerId = get()[centerKey(side)]
    if (!centerId) return
    const key = `neighborhood:${side}`
    const ticket = seq.next(key)
    set((s) => { s.loading[side] = true; s.error[side] = null })
    try {
      const data = await getNeighborhood({ centerId, hops: get().hops })
      if (!seq.isCurrent(key, ticket)) return          // 过期响应，丢弃
      set((s) => { s[subKey(side)] = data })
    } catch (e) {
      if (!seq.isCurrent(key, ticket)) return
      set((s) => { s.error[side] = (e as ApiError).message })
    } finally {
      if (seq.isCurrent(key, ticket)) set((s) => { s.loading[side] = false })
    }
  },

  loadPath: async () => {
    const { startId, endId } = get()
    if (!startId || !endId) { set((s) => { s.path = null }); return }
    const ticket = seq.next('path')
    set((s) => { s.loading.path = true })
    try {
      const data = await getPath({ from: startId, to: endId })
      if (!seq.isCurrent('path', ticket)) return
      set((s) => { s.path = data })
    } finally {
      if (seq.isCurrent('path', ticket)) set((s) => { s.loading.path = false })
    }
  },

  loadStats: async () => {
    const data = await getStats()
    set((s) => { s.stats = data })
  },

  // 写操作成功后调用：独立的读并行发出，不串成瀑布
  refreshAll: async () => {
    await Promise.all([
      get().loadSide('L'), get().loadSide('R'),
      get().loadPath(), get().loadStats(), get().reloadLinks(),
    ])
  },
})
```
